#!/usr/bin/env python3
"""Bounded synthetic Docker qualification; leaves its 64 MiB test volume.

Run with worker runtime configuration JSON; never accesses credentials or v1.
Only this invocation's randomized cwb2 runtime/network objects are cleaned up.
"""
from __future__ import annotations

raise SystemExit('Archived operation is disabled. See docs/AGENT-SETUP.md for supported installation.')


import argparse
import json
from pathlib import Path
import sys
import time
import uuid

from cloudworkbench.runtime import Runtime


ISOLATION = r'''

checks={}

checks['uid_distinct_from_personal']=os.getuid()!=PERSONAL_UID

checks['no_new_privileges']='NoNewPrivs:\t1' in pathlib.Path('/proc/self/status').read_text()
checks['capabilities_dropped']='CapEff:\t0000000000000000' in pathlib.Path('/proc/self/status').read_text()
try:
    pathlib.Path('/isolation-write-test').write_text('x')
    checks['root_write_denied']=False

    checks['root_write_denied']=exc.errno in (errno.EROFS,errno.EACCES)
checks['readonly_root']=any(line.split()[4]=='/' and 'ro' in line.split()[5].split(',') for line in pathlib.Path('/proc/self/mountinfo').read_text().splitlines())
checks['sudo_absent']=not pathlib.Path('/usr/bin/sudo').exists()

    checks['embedded_dns_denied']=len(response)<12 or response[3]&15 != 0 or response[6:8]==b'\x00\x00'

QUOTA = r'''
import errno,json,os,pathlib
path=pathlib.Path('/workspace/qualification-enospc.bin')
written=0;blocked=False;error=None
try:
    with path.open('xb') as output:
        for _ in range(80):
            output.write(b'x'*(1024*1024));output.flush();os.fsync(output.fileno());written+=1024*1024
except OSError as exc:
    blocked=exc.errno in (errno.ENOSPC,errno.EDQUOT);error=exc.errno
finally:
    if path.exists():path.unlink()
print(json.dumps({'checks':{'hard_quota_enforced':blocked,'bounded_write':written<=64*1024*1024},'written_before_failure':written,'errno':error}),flush=True)
'''

NETWORK = r'''
import json,os,socket,ssl,urllib.parse,http.client
checks={}
proxy=urllib.parse.urlsplit(os.environ['HTTPS_PROXY'])
for name,target in [('proxy_private','127.0.0.1:443'),('proxy_tailnet','100.64.0.10:443'),('proxy_metadata','169.254.169.254:443'),('proxy_ipv6_loopback','[::1]:443'),('proxy_unlisted','example.org:443')]:
    try:
        with socket.create_connection((proxy.hostname,proxy.port),2) as s:
            s.sendall(('CONNECT '+target+' HTTP/1.1\r\nHost: '+target+'\r\n\r\n').encode())
            response=s.recv(1024)
            checks[name+'_denied']=response.startswith(b'HTTP/1.1 403')
    except OSError:checks[name+'_denied']=False
try:
    conn=http.client.HTTPSConnection(proxy.hostname,proxy.port,timeout=15,context=ssl.create_default_context())
    conn.set_tunnel('github.com',443)
    conn.request('HEAD','/',headers={'Host':'github.com','User-Agent':'cloudworkbench-qualification'})
    response=conn.getresponse()
    checks['allowlisted_github_tls']=200<=response.status<400
    status=response.status;conn.close()
except Exception as exc:
    checks['allowlisted_github_tls']=False;status=type(exc).__name__
print(json.dumps({'checks':checks,'github_status':status}),flush=True)

def run_case(runtime: Runtime, session: str, tag: str, program: str, timeout: int = 60) -> dict:
    attempt = f'qual-{tag}-{uuid.uuid4().hex[:12]}'
    runtime_id = None
    report = {'attempt_id':attempt,'checks':{},'passed':False}
    try:
        runtime_id = runtime.launch(attempt,session,['python3','-c',program],{},generation=1)
        deadline=time.monotonic()+timeout
        status={'state':'unknown'}
        while time.monotonic()<deadline:
            status=runtime.status(runtime_id,expected_generation=1)
            if status['state']!='running':break
            time.sleep(.25)
        if status['state']=='running':
            runtime.stop(runtime_id,expected_generation=1)
            report['error']='synthetic_test_timeout'
        else:
            logs=runtime.logs(runtime_id,max_bytes=32768).decode('utf-8','replace')
            records=[]
            for line in logs.splitlines():
                try:
                    value=json.loads(line)
                    if isinstance(value,dict) and 'checks' in value:records.append(value)
                except ValueError:pass
            if len(records)!=1:
                report['error']='missing_or_ambiguous_test_receipt'
            else:
                report.update(records[0])
            report['exit_code']=status.get('exit_code')
            report['runtime_id']=runtime_id
            report['passed']=status.get('exit_code')==0 and bool(report['checks']) and all(v is True for v in report['checks'].values())
    except Exception as exc:
        report['error']=type(exc).__name__+': '+str(exc)[:200]
    finally:
        try:
            if runtime_id:
                runtime.stop(runtime_id,expected_generation=1)
                runtime.cleanup(runtime_id,expected_generation=1)
            else:
                # Only this invocation's attempt; no v1 resources or old paths.
                # Discover a timed-out create before touching its support objects.
                for item in runtime.list_owned():
                    if item['labels'].get('io.cloudworkbench.attempt')==attempt and item['labels'].get('io.cloudworkbench.generation')=='1':
                        runtime.stop(item['runtime_id'],expected_generation=1)
                        runtime.cleanup(item['runtime_id'],expected_generation=1)
                runtime.cleanup_infrastructure(attempt,expected_generation=1)
            report['cleanup']='owned_runtime_objects_removed'
        except Exception as exc:
            report['cleanup']='failed: '+type(exc).__name__
            report['passed']=False
    return report


def main() -> int:
    parser=argparse.ArgumentParser()
    parser.add_argument('--config',required=True,type=Path)
    parser.add_argument('--output',required=True,type=Path)
    parser.add_argument('--personal-uid',type=int,default=1000)
    args=parser.parse_args()
    config=json.loads(args.config.read_text())
    # Accept standalone runtime config or parent service config's runtime block.
    config=dict(config.get('runtime',config))
    config.update(workspace_mib=64,test_path_workspace=False,network_enabled=False)
    config.pop('approved_mount_roots',None)
    config.pop('approved_writable_mount_roots',None)
    config.pop('workspace_readonly',None)
    runtime=Runtime(config)
    session='qualification-'+uuid.uuid4().hex
    report={'schema_version':1,'session_id':session,'scope':'synthetic 64MiB qualification only',
            'left_volume_for_evidence':str(runtime.root/session),'cases':{},'passed':False}
    try:
        if runtime.uid==args.personal_uid or runtime.uid==0:
            raise ValueError('job UID must be distinct from personal UID and root')
        program=ISOLATION.replace('PERSONAL_UID',str(args.personal_uid)).replace('EXPECTED_UID',str(runtime.uid))
        report['cases']['no_network_isolation']=run_case(runtime,session,'none',program)
        report['cases']['quota']=run_case(runtime,session,'quota',QUOTA)
        if 'github.com' in config.get('allowed_domains',[]):
            network=Runtime({**config,'network_enabled':True})
            report['cases']['isolated_gateway_isolation']=run_case(network,session,'isolated',program)
            report['cases']['proxy_policy_and_tls']=run_case(network,session,'proxy',NETWORK)
        else:
            report['cases']['proxy_policy_and_tls']={'passed':False,'checks':{},'skipped':'github.com not explicitly configured; no allowlist expansion'}
        report['passed']=all(case.get('passed') is True for case in report['cases'].values())
    except Exception as exc:
        report['error']=type(exc).__name__+': '+str(exc)[:200]
    args.output.parent.mkdir(parents=True,exist_ok=True)
    # Never overwrite an existing evidence file.
    with args.output.open('x') as output:json.dump(report,output,indent=2)
    print(json.dumps(report,indent=2))
    return 0 if report['passed'] else 1


if __name__=='__main__':
    sys.exit(main())
