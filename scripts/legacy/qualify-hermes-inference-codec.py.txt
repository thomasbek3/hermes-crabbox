"""No-network SDK wire serialization and native Hermes normalizer/accumulator proof."""

# Archived one-off operation; use the supported host installer instead.
raise SystemExit('Archived operation is disabled. See docs/AGENT-SETUP.md for supported installation.')

import hashlib,json,os
from pathlib import Path
import httpx
from openai import OpenAI
from openai.types.chat import ChatCompletion,ChatCompletionChunk
from agent.transports.chat_completions import ChatCompletionsTransport
from agent.chat_completion_helpers import _ToolCallAccumulator
from cloudworkbench.inference_transport import PinnedCLI,InferenceResult
from cloudworkbench.hermes_inference_protocol import admit_request,encode_response

stage=Path(__file__).resolve().parents[2]
profile=PinnedCLI(Path('/synthetic/claude'),'a'*64,'2.1.274','claude-opus-4-6','high')
schemas=json.loads((stage/'tests/fixtures/hermes/base-tool-schemas.json').read_text())
print_type=type(schemas)
if isinstance(schemas,dict):
    schemas=schemas.get('tools',schemas)
assert isinstance(schemas,list),print_type
schemas=[s for s in schemas if s['function']['name'] in {'read_file','write_file','patch','search_files'}]
assert len(schemas)==4
transport=ChatCompletionsTransport()
captured=[];checks=[]
placeholder={'id':'chatcmpl-placeholder','object':'chat.completion','created':1,'model':profile.native_model,'choices':[{'index':0,'message':{'role':'assistant','content':'placeholder'},'finish_reason':'stop'}]}
def capture(request):
    captured.append(json.loads(request.content))
    return httpx.Response(200,json=placeholder)
client=OpenAI(api_key='synthetic-no-provider',base_url='http://synthetic.invalid/v1',http_client=httpx.Client(transport=httpx.MockTransport(capture)),max_retries=0)
messages=[{'role':'system','content':'File-only synthetic qualification.'},{'role':'user','content':'Read a.txt.'}]
kwargs=transport.build_kwargs(profile.native_model,messages,schemas,max_tokens=None,reasoning_config={'enabled':True,'effort':'high'},supports_reasoning=False,base_url='http://synthetic.invalid/v1',timeout=2)
client.chat.completions.create(**kwargs)
admitted=admit_request(captured[-1],profile)
assert 'timeout' not in captured[-1] and 'max_tokens' not in captured[-1]
checks.append('native_wire_admitted')
identity={'model':profile.native_model,'effort':profile.effort,'cli_version':profile.version,'binary_sha256':profile.sha256,'trust':'controller_pinned'}
decision={'kind':'tool_calls','text':None,'tool_calls':[{'id':'call_native_1','name':'read_file','arguments':{'path':'a.txt'}}]}
result=InferenceResult('ok',decision,None,identity,{'model':None,'effort':None,'status':'unknown'},None,{'state':'unknown'},None,'unknown','worker_reported',{'scope':'process_group','process_group_stopped':True,'container_cleanup_required':True,'credential_reuse_authorized':False})
response=encode_response(admitted,result,completion_id='chatcmpl-native',created=1)
normalized=transport.normalize_response(ChatCompletion.model_validate_json(response.body))
assert normalized.tool_calls[0].id=='call_native_1' and json.loads(normalized.tool_calls[0].arguments)=={'path':'a.txt'}
assert normalized.usage is None
checks.append('native_normalizer_preserves_unknown_usage_and_call')
# Real SDK serializes streaming kwargs; mock HTTP has no socket or provider.
kwargs.update(stream=True,stream_options={'include_usage':True})
stream=client.chat.completions.create(**kwargs);stream.close()
streamed=encode_response(admit_request(captured[-1],profile),result,completion_id='chatcmpl-native',created=1)
accumulator=_ToolCallAccumulator();finishes=[]
for frame in streamed.body.decode().split('\n\n'):
    if not frame or frame=='data: [DONE]':continue
    chunk=ChatCompletionChunk.model_validate(json.loads(frame[6:]))
    for choice in chunk.choices:
        for call in choice.delta.tool_calls or []:accumulator.feed(call)
        if choice.finish_reason:finishes.append(choice.finish_reason)
materialized=accumulator.materialize()
assert materialized[0]['id']=='call_native_1' and json.loads(materialized[0]['function']['arguments'])=={'path':'a.txt'}
assert finishes==['tool_calls']
checks.append('native_stream_accumulator_preserves_call')
# Actual native conversion strips persistence-only tool sidecars while preserving correlation.
messages.extend([{'role':'assistant','content':None,'tool_calls':[{'id':'call_native_1','type':'function','function':{'name':'read_file','arguments':'{"path":"a.txt"}'}}]}, {'role':'tool','name':'read_file','tool_call_id':'call_native_1','content':'synthetic file'}])
client.chat.completions.create(**transport.build_kwargs(profile.native_model,messages,schemas,max_tokens=None,timeout=2))
roundtrip=admit_request(captured[-1],profile)
assert roundtrip.transport_request['messages'][-1]=={'role':'tool','tool_call_id':'call_native_1','content':'synthetic file'}
checks.append('native_tool_result_correlation')
client.close()
print(json.dumps({'passed':len(checks),'checks':checks,'failed':0,'suite':'native Hermes wire/nonstream/SSE/tool-correlation','source_sha256':json.loads((stage/'source-manifest.json').read_text()),'wire_requests':captured,'network_calls':False,'provider_calls':False,'usage_unknown_preserved':True,'credential_reuse_authorized':False}))
