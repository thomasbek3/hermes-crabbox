#!/usr/bin/env python3
"""Publish the authorized Crabbox Hermes lane without running qualification jobs."""
from __future__ import annotations

raise SystemExit('Archived one-off activation is disabled; use docs/AGENT-SETUP.md.')


import argparse
import copy
import hashlib
import json
import os
from pathlib import Path
import re
import runpy
import socket
import sqlite3
import stat
import sys
import time


VERSION = 'hermes-tasks-desktop-v1'
PREVIOUS_VERSION = 'hermes-tasks-crabbox-v1'
PROFILE = 'hermes-crabbox-bridge'
CRABBOX = Path('/opt/crabbox/0.61.0/crabbox')
MODULES = ('crabbox_runtime.py', 'crabbox_guest.py', 'desktop_bridge.py')
NEW_MODULES = frozenset({'desktop_bridge.py'})


def require(condition, code):
    if not condition:
        raise RuntimeError(code)


def protected_file(path):
    info = path.lstat()
    require(stat.S_ISREG(info.st_mode) and info.st_nlink == 1 and info.st_uid == 0 and
            not info.st_mode & 0o022 and path.resolve() == path, 'protected_file_required')


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--stage', type=Path, required=True)
    parser.add_argument('--image', required=True)
    args = parser.parse_args()
    require(os.geteuid() == 0 and socket.gethostname() == 'archived-worker.invalid', 'root_omarchy_required')
    root = Path('/var/lib/cloud-workbench')
    require(args.stage.is_absolute() and args.stage.is_relative_to(root / 'qualifications') and
            args.stage.resolve() == args.stage and args.stage.is_dir() and
            args.stage.stat().st_uid == 0 and not args.stage.stat().st_mode & 0o022,
            'protected_stage_required')
    helper = args.stage / 'activate-basic-delegation.py'
    protected_file(helper)
    helpers = runpy.run_path(str(helper))
    run, atomic, encoded, idle = (helpers[key] for key in ('run', 'atomic', 'encoded', 'idle'))
    package = helpers['PACKAGE']
    require(helpers['ROOT'] == root, 'helper_root_mismatch')
    require(re.fullmatch(r'sha256:[0-9a-f]{64}', args.image), 'immutable_image_required')
    protected_file(CRABBOX)
    require(os.access(CRABBOX, os.X_OK), 'crabbox_not_executable')
    inspected = json.loads(run('/usr/bin/docker', 'image', 'inspect', args.image))[0]
    require(inspected['Id'] == args.image and inspected['Os'] == 'linux' and
            inspected['Architecture'] == 'amd64', 'image_identity_mismatch')
    binding_path = args.stage / 'source-binding.json'
    protected_file(binding_path)
    bindings = json.loads(binding_path.read_text())
    sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
    require(type(bindings) is dict and type(bindings.get('before')) is dict and
            type(bindings.get('after')) is dict and set(bindings['before']) == set(MODULES) and
            set(bindings['after']) == set(MODULES), 'invalid_source_binding')

    def check_source(name):
        protected_file(args.stage / name)
        require(type(bindings['after'][name]) is str and
                re.fullmatch(r'[0-9a-f]{64}', bindings['after'][name]) and
                sha(args.stage / name) == bindings['after'][name], 'staged_source_drift')
        before = bindings['before'][name]
        if before is None:
            require(name in NEW_MODULES and not os.path.lexists(package / name), 'source_drift')
        else:
            require(type(before) is str and re.fullmatch(r'[0-9a-f]{64}', before), 'invalid_source_binding')
            protected_file(package / name)
            require(sha(package / name) == before, 'source_drift')

    for name in MODULES:
        check_source(name)
    config_paths = {name: Path('/etc/cloud-workbench') / (name + '.json')
                    for name in ('api', 'worker')}
    for path in config_paths.values():
        protected_file(path)
    originals = {name: path.read_bytes() for name, path in config_paths.items()}
    configs = {name: json.loads(raw) for name, raw in originals.items()}
    database = root / 'control/state.db'
    registry_path = root / 'environments' / (VERSION + '-attempt2.db')
    backup = root / 'operator-backups/crabbox-desktop-20260918-attempt2'
    require(not os.path.lexists(backup) and not os.path.lexists(registry_path),
            'crabbox_activation_already_attempted')
    idle(database)
    legacy = run('systemctl', 'show', 'cloudd', '--property=MainPID', '--value')
    backup.mkdir(mode=0o700)
    receipt = {
        'host': 'archived-worker.invalid', 'started_at': time.time(), 'stage': 'preparing', 'new_job_runs': 0,
        'verification_runs': 0, 'image': args.image, 'environment_version': VERSION,
        'environment_qualified': False,
        'authorization': 'The operator requested graphical desktop and VNC for Omarchy workers',
        'source_sha256': bindings['after'], 'previous_source_sha256': bindings['before'],
        'new_source_files': [name for name in MODULES if bindings['before'][name] is None],
        'backup': str(backup),
    }

    def record():
        atomic(backup / 'deployment.json', encoded(receipt))

    record()
    try:
        for service in ('cloud-workbench-api', 'cloud-workbench-worker'):
            run('systemctl', 'stop', service)
        idle(database)
        for name, path in config_paths.items():
            require(path.read_bytes() == originals[name], 'config_drift')
            atomic(backup / (name + '.json'), originals[name])
        for name in MODULES:
            check_source(name)
            if bindings['before'][name] is not None:
                atomic(backup / name, (package / name).read_bytes())
        atomic(backup / 'source-binding.json', encoded(bindings))
        with sqlite3.connect(database.as_uri() + '?mode=ro', uri=True) as source, \
                sqlite3.connect(backup / 'state.db') as saved:
            source.backup(saved)
        (backup / 'state.db').chmod(0o600)
        previous = Path(configs['worker']['environment_registry'])
        require(configs['api']['environment_registry'] == str(previous), 'registry_config_mismatch')
        with sqlite3.connect(previous.as_uri() + '?mode=ro', uri=True) as source, \
                sqlite3.connect(registry_path) as copied:
            source.backup(copied)
        from cloudworkbench.environments import EnvironmentRegistry
        registry = EnvironmentRegistry(registry_path)
        approval = configs['worker']['projects']['hermes-tasks'].get(
            'operator_approved_environments', {}).get(PREVIOUS_VERSION)
        require(type(approval) is str and configs['api']['projects']['hermes-tasks'].get(
            'operator_approved_environments', {}).get(PREVIOUS_VERSION) == approval,
            'previous_environment_approval_mismatch')
        old = registry.resolve('hermes-tasks', PREVIOUS_VERSION, operator_approved_sha256=approval)
        manifest = copy.deepcopy(old['manifest'])
        manifest.update(version=VERSION, image_digest=args.image, network_profile=PROFILE,
                        base_image_digest=old['manifest']['image_digest'])
        manifest['resources'] = {'cpus': 1, 'memory_mib': 4096, 'pids': 512, 'workspace_mib': 8192}
        manifest['cli_versions']['crabbox'] = '0.61.0'
        candidate = registry.register(manifest)
        require(candidate['qualified'] is False and candidate['qualification'] is None,
                'unexpected_qualification')
        receipt['retired_submission_profile'] = PREVIOUS_VERSION
        receipt['retired_reason'] = 'Previous untested image lacks per-container SSH host-key generation; desktop image fixes startup'
        receipt['manifest_sha256'] = candidate['manifest_sha256']
        receipt['environment'] = candidate
        for config in configs.values():
            project = config['projects']['hermes-tasks']
            require(VERSION not in project['environment_versions'], 'environment_already_configured')
            project['environment_versions'].append(VERSION)
            project['environment_versions'].remove(PREVIOUS_VERSION)
            project.setdefault('operator_approved_environments', {})[VERSION] = candidate['manifest_sha256']
            project.setdefault('environment_resources', {})[VERSION] = copy.deepcopy(manifest['resources'])
            config['environment_registry'] = str(registry_path)
        worker = configs['worker']
        worker['operator_approved_images'] = list(dict.fromkeys(
            worker.get('operator_approved_images', []) + [args.image]))
        policy = worker['hermes_runtime']
        policy.update(crabbox_enabled=True, crabbox_binary=str(CRABBOX))
        policy['crabbox_images'] = list(dict.fromkeys(policy.get('crabbox_images', [policy['crabbox_image']]) + [args.image]))
        policy['crabbox_desktop_images'] = list(dict.fromkeys(policy.get('crabbox_desktop_images', []) + [args.image]))
        profiles = list(policy.get('environment_network_profiles', []))
        if policy.get('environment_network_profile'):
            profiles.append(policy['environment_network_profile'])
        policy['environment_network_profiles'] = list(dict.fromkeys(profiles + [PROFILE]))
        policy.setdefault('tool_network_by_profile', {})[PROFILE] = 'bridge'
        policy['tool_image_allowlist'] = list(dict.fromkeys(
            policy.get('tool_image_allowlist', []) + [old['manifest']['image_digest'], args.image]))
        controller = root / 'worker/crabbox-runtime'
        if os.path.lexists(controller):
            info = controller.lstat()
            require(stat.S_ISDIR(info.st_mode) and controller.resolve() == controller and
                    info.st_uid == 959 and info.st_gid == 960 and
                    stat.S_IMODE(info.st_mode) in (0o700, 0o2700), 'unsafe_crabbox_runtime_directory')
            receipt['runtime_directory_created'] = False
        else:
            require(controller.parent.resolve() == controller.parent, 'unsafe_worker_directory')
            controller.mkdir(mode=0o700)
            os.chown(controller, 959, 960)
            receipt['runtime_directory_created'] = True
        receipt['stage'] = 'publication'
        record()
        for name in MODULES:
            check_source(name)
            atomic(package / name, (args.stage / name).read_bytes(), 0o644)
        os.chown(registry_path, 0, 960)
        os.chmod(registry_path, 0o640)
        atomic(registry_path.with_suffix('.approval.json'), encoded({
            'manifest_sha256': candidate['manifest_sha256'], 'image': args.image,
            'operator_approved': True, 'qualified': False, 'tests_run': 0, 'profile': PROFILE,
        }), 0o640, 960)
        for name, path in config_paths.items():
            atomic(path, encoded(configs[name]), 0o640, 960)
        for service in ('cloud-workbench-worker', 'cloud-workbench-api'):
            run('systemctl', 'start', service)
        receipt['services'] = {service: run('systemctl', 'is-active', service)
                               for service in ('cloud-workbench-worker', 'cloud-workbench-api')}
        require(run('systemctl', 'show', 'cloudd', '--property=MainPID', '--value') == legacy,
                'legacy_service_changed')
        receipt['stage'] = 'activated'
    except Exception as exc:
        receipt['error'] = str(exc) if type(exc) is RuntimeError else type(exc).__name__
        receipt['recovery'] = 'Inspect retained backup and partial publication; no blind rerun or task database rollback.'
        raise
    finally:
        receipt['finished_at'] = time.time()
        record()
    print(json.dumps({key: value for key, value in receipt.items() if key != 'environment'}, indent=2))


if __name__ == '__main__':
    try:
        main()
    except Exception as exc:
        print(json.dumps({'activated': False,
                          'error': str(exc) if type(exc) is RuntimeError else type(exc).__name__}))
        sys.exit(1)
