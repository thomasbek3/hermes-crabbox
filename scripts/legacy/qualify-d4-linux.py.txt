def command(args,timeout=30):
    result=subprocess.run(args,text=True,capture_output=True,timeout=timeout)
    if result.returncode:
        raise RuntimeError('command_failed:'+args[0]+':'+str(result.returncode))
    return result.stdout

def inspect(identity):

    labels=value['Config']['Labels']
    assert labels.get('cwb.proof.owner')==OWNER and labels.get('cwb.proof.run')==RUN and value['Image']==IMAGE
    return value

        '--mount','type=bind,src='+str(source)+',dst=/proof,readonly',
        '--env','PYTHONPATH=/proof/source','--env','PYTHONDONTWRITEBYTECODE=1',
        '--entrypoint','/opt/hermes/venv/bin/python']
    if role_mount is not None:
        args += ['--mount','type=bind,src='+str(role)+',dst=/run/cloud-role-broker'+(',readonly' if role_mount=='ro' else '')]
