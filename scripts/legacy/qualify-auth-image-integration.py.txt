"""Post-rollout real-provider image-pinning and reversible retention qualification."""

from datetime import datetime,timezone
import argparse,hashlib,json,os
from pathlib import Path
import runpy,socket,subprocess,sys,tempfile,time,urllib.error,urllib.request,uuid
from cloudworkbench.repositories import snapshot,initialize_workspace

sha=lambda data:hashlib.sha256(data).hexdigest()
now=lambda:datetime.now(timezone.utc).isoformat()

def event_history(sid):
    data,_=call('GET',f'/v1/sessions/{sid}/events?follow=false',raw=True)
    return [json.loads(line[5:]) for line in data.decode().splitlines() if line.startswith('data:')]

def sse(sid,cursor=0,stop_on_tool=False):
    headers={'Authorization':'Bearer '+TOKEN,'Accept':'text/event-stream'}
    if cursor:headers['Last-Event-ID']=str(cursor)
    request=urllib.request.Request(BASE+f'/v1/sessions/{sid}/events?follow=true',headers=headers)
    start=time.monotonic();events=[];block=[];was_active=False
    with urllib.request.urlopen(request,timeout=20) as stream:
        assert stream.headers.get_content_type()=='text/event-stream'
        while time.monotonic()-start<300:
            line=stream.readline(1024*1024)
            if not line:break
            assert len(line)<1024*1024
            line=line.decode().rstrip('\r\n')
            if line:
                if not line.startswith(':'):block.append(line)
                continue
            data=next((line[5:].strip() for line in block if line.startswith('data:')),None)
            eventid=next((line[3:].strip() for line in block if line.startswith('id:')),None)
            block=[]
            if data is None:continue
            event=json.loads(data)
            assert int(eventid)==event['sequence'] and event['sequence']>cursor
            cursor=event['sequence'];events.append(event)
            state=call('GET','/v1/sessions/'+sid)['state']
            was_active=was_active or state not in {'completed','failed','cancelled','interrupted'}
            if stop_on_tool and event['type']=='tool.started':
                assert state not in {'completed','failed','cancelled','interrupted'}
                return events,{'last_event_id':cursor,'disconnected_while_active':True,'event_count':len(events)}
    return events,{'last_event_id':cursor,'observed_active':was_active,'event_count':len(events)}

def wait(sid,turns):
    deadline=time.monotonic()+600
    while time.monotonic()<deadline:
        result=call('GET','/v1/sessions/'+sid)
        if len(result.get('turns',[]))>=turns and result['state'] in {'completed','failed','cancelled','interrupted'}:return result
        time.sleep(.5)
    raise RuntimeError('bounded task wait expired')

def verify_delivery(sid,attempt_id):
    result=call('GET',f'/v1/sessions/{sid}/diff')
    assert result['attempt_id']==attempt_id and result['base_commit']==definition['commit']
    assert result['complete_text_patch'] and result['excluded_from_text_patch']==[]
    artifacts=call('GET',f'/v1/sessions/{sid}/artifacts')['artifacts']
    bodies={};metadata=[];modes={}
    for item in artifacts:
        if item['attempt_id']!=attempt_id:continue
        content,headers=call('GET',f'/v1/artifacts/{item["id"]}/content',raw=True)
        assert sha(content)==item['sha256'] and len(content)==item['bytes']
        assert headers.get('content-disposition','').startswith('attachment')
        modes[item['path']]=item['executable']
        bodies[item['path']]=content;metadata.append({key:item[key] for key in ('path','sha256','bytes','id','executable')})
    patch=bodies['@delivery/changes.patch'];assert sha(patch)==result['patch_sha256']
    manifest=json.loads(bodies['@delivery/manifest.json']);assert manifest['changed_files']==result['changed_files']
    baseline=snapshot(Path(registered['path']),definition['commit'])
    env={'PATH':'/usr/bin:/bin','HOME':'/nonexistent','GIT_CONFIG_NOSYSTEM':'1','GIT_CONFIG_GLOBAL':'/dev/null','GIT_TERMINAL_PROMPT':'0'}
    with tempfile.TemporaryDirectory(prefix='cwb-patch-proof-') as temporary:
        root=Path(temporary);work=root/'work';work.mkdir();initialize_workspace(work,baseline)
        patch_path=root/'changes.patch';patch_path.write_bytes(patch)
        # No agent Git configuration or hook can enter this independently constructed tree.
        command=['git','-c','core.hooksPath=/dev/null','-C',str(work),'apply']
        subprocess.run(command+['--check',str(patch_path)],env=env,check=True,capture_output=True,timeout=30)
        subprocess.run(command+[str(patch_path)],env=env,check=True,capture_output=True,timeout=30)
        for changed in result['changed_files']:
            path=work/changed['path']
            if changed['status']=='deleted':assert not path.exists()
            else:
                assert sha(path.read_bytes())==changed['after_sha256']==sha(bodies[changed['path']])
                assert changed['after_mode'] in ('100644','100755')
                executable=changed['after_mode']=='100755'
                assert bool(path.stat().st_mode & 0o100)==executable==modes[changed['path']]
    return {'diff':result,'artifacts':metadata,'patch_apply_check':True,'applied_hashes_match':True,'applied_modes_match':True,'verification_source':'protected worker runtime; patch bytes matched exported artifact hashes'}


def main(argv=None):
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--deployment-receipt',type=Path,required=True)
    parser.add_argument('--legacy-session',default='eb6fc68d-8087-4bd8-864b-cd6f1ec8be19')
    parser.add_argument('--execute',action='store_true')
    args=parser.parse_args(argv)
    if not args.execute:parser.error('Pass --execute only after the reviewed rollout and readiness checks')
    if socket.gethostname()!='archived-worker.invalid':raise ValueError('Wrong host')
    global call,BASE,TOKEN,definition,registered
    helper=runpy.run_path(str(Path(__file__).with_name('http-smoke.py')))
    call,BASE,TOKEN=helper['call'],helper['BASE'],helper['TOKEN']
    deployment=json.loads(args.deployment_receipt.read_text())
    assert deployment['passed'] and deployment['execute'] and deployment['schema_version']==1
    assert deployment['host']==socket.gethostname()=='archived-worker.invalid'
    started=datetime.fromisoformat(deployment['started_at']);finished=datetime.fromisoformat(deployment['finished_at'])
    assert started.tzinfo is not None and finished.tzinfo is not None
    assert 0 <= (finished-started).total_seconds() <= 7200
    assert 0 <= (datetime.now(timezone.utc)-finished).total_seconds() <= 86400
    config=json.loads(Path('/etc/cloud-workbench/worker.json').read_text())
    assert config['runtime']['image']==deployment['new_image_id']
    assert sha(Path('/etc/cloud-workbench/worker.json').read_bytes())==deployment['config_sha256_after']['worker']
    assert sha(Path('/etc/cloud-workbench/api.json').read_bytes())==deployment['config_sha256_after']['api']
    for name,digest in deployment['deployed_source_sha256'].items():
        assert sha((Path('/opt/cloud-workbench')/name).read_bytes())==digest
    definition=config['projects']['sample-repo']['repository'];registered=config['repositories'][definition['repository_id']]
    assert call('GET','/v1/ready')['provider_enabled']
    legacy=call('GET','/v1/sessions/'+args.legacy_session)
    assert legacy['state']=='completed' and legacy['project_id']=='sample-repo'
    old_pin=legacy['attempts'][-1]['result']['environment']
    assert old_pin['manifest']['image_digest']==deployment['old_image_id']
    receipt={'schema_version':1,'started_at':now(),'host':socket.gethostname(),'driver_sha256':sha(Path(__file__).read_bytes()),
        'deployment_receipt_sha256':sha(args.deployment_receipt.read_bytes()),'old_image_id':deployment['old_image_id'],
        'new_image_id':deployment['new_image_id'],'provider_called':True,'passed':False,'checks':{}}
    restore=None
    try:
        created=call('POST','/v1/sessions',{'project_id':'sample-repo','agent':'claude','environment_version':'repo-v2',
            'goal':'Fix the supplied booking.py valid_date(value) to accept only real calendar dates exactly YYYY-MM-DD, rejecting malformed strings, impossible dates and nonstrings. Add unit tests and run them. Write report.md explaining the fix and results. Use the local repository only; no dependency installation or network fetching.',
            'acceptance':[{'id':'booking-validity','description':'Valid dates pass; malformed and nonexistent dates fail.','mandatory':True}]})
        sid=receipt['new_session_id']=created['session_id']
        events,sse_receipt=sse(sid);result=wait(sid,1)
        assert result['state']=='completed' and result['outcome']=='verified'
        first=next(attempt for attempt in result['attempts'] if attempt['id']==created['attempt_id'])
        assert first['result']['image_digest']==deployment['new_image_id']
        new_pin=first['result']['environment'];assert new_pin['manifest']['version']=='repo-v2'
        assert new_pin['manifest']['image_digest']==deployment['new_image_id']
        receipt['new_first']={'attempt_id':first['id'],'state':first['state'],'outcome':first['outcome'],'image_digest':first['result']['image_digest'],'manifest_sha256':new_pin['manifest_sha256'],'sse':sse_receipt,'delivery':verify_delivery(sid,first['id'])}
        receipt['checks']['new_image_real_provider_and_verifier']=True
        history=event_history(sid)
        follow=call('POST',f'/v1/sessions/{sid}/messages',{'message':'Update README.md with accepted/rejected date examples and the exact unit test command. Preserve the implementation/report and rerun tests.'})
        _,follow_sse=sse(sid,cursor=history[-1]['sequence']);result=wait(sid,2)
        follow_attempt=next(a for a in result['attempts'] if a['id']==follow['attempt_id'])
        assert follow_attempt['state']=='completed' and follow_attempt['outcome']=='verified'
        assert follow_attempt['result']['environment']==new_pin
        assert follow_attempt['result']['image_digest']==deployment['new_image_id']
        receipt['new_followup']={'attempt_id':follow_attempt['id'],'image_digest':follow_attempt['result']['image_digest'],'sse':follow_sse,'delivery':verify_delivery(sid,follow_attempt['id'])}
        receipt['checks']['new_followup_pinned']=True
        old_cursor=event_history(args.legacy_session)[-1]['sequence']
        old_follow=call('POST',f'/v1/sessions/{args.legacy_session}/messages',{'message':'Verify the existing local booking unit tests still pass. Append one short validation note to report.md. Preserve the current implementation and other files.'})
        _,old_sse=sse(args.legacy_session,cursor=old_cursor)
        old_result=wait(args.legacy_session,len(legacy['turns'])+1)
        old_attempt=next(a for a in old_result['attempts'] if a['id']==old_follow['attempt_id'])
        assert old_attempt['state']=='completed' and old_attempt['outcome']=='verified'
        assert old_attempt['result']['environment']==old_pin
        assert old_attempt['result']['image_digest']==deployment['old_image_id']
        receipt['old_followup']={'session_id':args.legacy_session,'attempt_id':old_attempt['id'],'image_digest':old_attempt['result']['image_digest'],'sse':old_sse,'delivery':verify_delivery(args.legacy_session,old_attempt['id'])}
        receipt['checks']['old_session_retains_old_image']=True
        path=f'/v1/sessions/{sid}/retention'
        before=call('GET',path);revision=before['policy']['revision']
        keep=before['policy']['keep'];key=str(uuid.uuid4())
        changed=call('PATCH',path,{'expected_revision':revision,'keep':True},key=key)
        restore=(path,changed['revision'],keep)
        assert call('PATCH',path,{'expected_revision':revision,'keep':True},key=key)==changed
        try:call('PATCH',path,{'expected_revision':revision,'keep':False})
        except urllib.error.HTTPError as error:assert error.code==409
        else:raise AssertionError('Stale retention revision accepted')
        dry=call('GET',path+'/dry-run')
        assert dry['policy']['keep'] and dry['deletion_manifest']==[]
        restored=call('PATCH',path,{'expected_revision':changed['revision'],'keep':keep});restore=None
        after=call('GET',path)
        assert after['policy']['keep']==keep
        receipt['retention']={'original_revision':revision,'changed_revision':changed['revision'],'restored_revision':restored['revision'],'keep_restored':True,'stale_revision_rejected':True,'idempotent_replay':True,'dry_run_empty_when_kept':True,'deletion_executed':False}
        receipt['checks']['reversible_retention_controls']=True
        receipt['passed']=True
    except Exception as error:
        receipt['error_type']=type(error).__name__
    finally:
        if restore:
            path,revision,keep=restore
            try:
                call('PATCH',path,{'expected_revision':revision,'keep':keep})
                receipt['retention_cleanup_restored']=True
            except Exception:receipt['retention_cleanup_restored']=False
        receipt['finished_at']=now()
        print(json.dumps(receipt,indent=2),flush=True)
    return 0 if receipt['passed'] else 1

if __name__=='__main__':raise SystemExit(main())
