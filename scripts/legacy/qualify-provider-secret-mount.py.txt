#!/usr/bin/env python3
"""Qualify synthetic readonly secret mounts without provider or network access."""

# Archived one-off operation; use the supported host installer instead.
raise SystemExit('Archived operation is disabled. See docs/AGENT-SETUP.md for supported installation.')

import datetime, hashlib, json, pathlib, shlex, subprocess, uuid
root=pathlib.Path(__file__).resolve().parents[2]
source=(root/'src/cloudworkbench/adapters.py').read_text()
expected=json.loads((root/'evidence/provider-bootstrap-binding.json').read_text())['files']['src/cloudworkbench/adapters.py']
assert hashlib.sha256(source.encode()).hexdigest()==expected
owner='secret-mount-'+uuid.uuid4().hex
image='sha256:5a03c9d2d1fc20683029c47683a3b0470ad2d7ce79eeb43ced1bdfba10770693'
remote=r'''
import os,tempfile,pathlib,json,subprocess,hashlib,socket
source=SOURCE
owner=OWNER
image=IMAGE
assert socket.gethostname()=='archived-worker.invalid'
folder=pathlib.Path(tempfile.mkdtemp(prefix='cwb-'+owner+'-'))
token=folder/'synthetic-token'
token.write_text('sk-ant-oat-synthetic-mount-proof-only')
checks=[]
cleanup=[]
try:
 for label,uid,mode,expected in [('foreign_owner',0,0o400,'auth_invalid'),('matching_owner',1000,0o400,None),('readonly_control',1000,0o600,None),('approved_group',0,0o640,None)]:
  os.chown(token,uid,1000);os.chmod(token,mode)
  name='cwb-'+owner+'-'+label
  program=source+"\nimport os,json,pathlib,errno\n"+"\np=pathlib.Path('/run/secrets/claude-token')\nopen_errno=None\ntry:\n fd=os.open(p,os.O_RDONLY)\nexcept OSError as exc:\n open_errno=exc.errno\nelse:\n os.close(fd)\nif p.stat().st_uid==0 and p.stat().st_mode&0o777==0o400: assert open_errno==errno.EACCES\nraw,error=read_credential(p)\nassert error=="+repr(expected)+"\nassert bool(raw)=="+repr(expected is None)+"\ntry:\n p.write_text('changed')\nexcept OSError as exc:\n denied=True\n write_errno=exc.errno\nelse:\n denied=False\nassert denied\ntry:\n os.chmod(p,0o600)\nexcept OSError as exc:\n chmod_errno=exc.errno\nelse:\n raise AssertionError('chmod unexpectedly allowed')\nif p.stat().st_uid==os.getuid(): assert chmod_errno==errno.EROFS\nif p.stat().st_mode&0o200: assert write_errno==errno.EROFS\nprint(json.dumps({'uid':os.getuid(),'secret_uid':p.stat().st_uid,'secret_gid':p.stat().st_gid,'mode':oct(p.stat().st_mode&0o777),'error':error,'write_denied':denied,'write_errno':write_errno,'open_errno':open_errno,'chmod_errno':chmod_errno,'accepted':bool(raw)}))\n"
  argv=['docker','run','--name',name,'--label','io.cloudworkbench.probe-owner='+owner,'--network','none','--read-only','--user','1000:1000','--cap-drop','ALL','--security-opt','no-new-privileges','--memory','128m','--pids-limit','32','--mount','type=bind,src='+str(token)+',dst=/run/secrets/claude-token,readonly','--entrypoint','/opt/hermes/venv/bin/python','-i',image,'-']
  try:
   result=subprocess.run(argv,input=program,text=True,capture_output=True,timeout=25)
   meta=json.loads(subprocess.check_output(['docker','inspect',name],text=True))[0]
   assert meta['Config']['Labels']['io.cloudworkbench.probe-owner']==owner
   assert meta['Image']==image
   mounts=meta['Mounts'];assert len(mounts)==1 and not mounts[0]['RW'] and mounts[0]['Destination']=='/run/secrets/claude-token'
   assert result.returncode==0, ('fixture failed',label,result.stderr[-1000:])
   checks.append({'case':label,'passed':True,'result':json.loads(result.stdout),'mounts':mounts,'image':meta['Image']})
  finally:
   inspection=subprocess.run(['docker','inspect',name],capture_output=True,text=True)
   if inspection.returncode==0:
    m=json.loads(inspection.stdout)[0]
    assert m['Config']['Labels']['io.cloudworkbench.probe-owner']==owner
    subprocess.run(['docker','rm','-f',name],check=True,capture_output=True)
   remaining=subprocess.check_output(['docker','ps','-aq','--filter','label=io.cloudworkbench.probe-owner='+owner],text=True).strip()
   assert not remaining
   cleanup.append({'case':label,'remaining':[]})
finally:
 token.unlink(missing_ok=True);folder.rmdir()
print(json.dumps({'host':socket.gethostname(),'checks':checks,'cleanup':cleanup,'tempfile_removed':not folder.exists(),'provider_calls':False,'real_credentials':False,'source_sha256':hashlib.sha256(source.encode()).hexdigest()}))
'''.replace('SOURCE',repr(source)).replace('OWNER',repr(owner)).replace('IMAGE',repr(image))
result=subprocess.run(['ssh','-o','BatchMode=yes','-o','ConnectTimeout=10','operator@worker.example.invalid','sudo python3 -'],input=remote,text=True,capture_output=True,timeout=100)
stamp=datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%SZ')
base=root/'evidence'/('provider-secret-mount-'+stamp)
base.with_suffix('.stderr.txt').write_text(result.stderr)
base.with_suffix('.json' if result.stdout.strip() else '.failed.txt').write_text(result.stdout)
assert result.returncode==0, result.stderr[-1500:]
data=json.loads(result.stdout);assert len(data['checks'])==4 and data['tempfile_removed'] and all(not c['remaining'] for c in data['cleanup'])
print(json.dumps({'receipt':str(base.with_suffix('.json')),'passed':4,'cleanup':data['cleanup']}))
