#!/usr/bin/env python3
"""Enable the requested Chromium/web profile without asserting unrun qualification."""
from __future__ import annotations

raise SystemExit('Archived one-off activation is disabled; use docs/AGENT-SETUP.md.')

import argparse
import copy
import hashlib
import json
import os
from pathlib import Path
import re
import runpy
import socket
import sqlite3
import sys
import time

VERSION = 'hermes-tasks-web-v2'
PROFILE = 'hermes-coordinator-bridge-tools-bridge'
MODULES = ('api.py', 'runner.py', 'environments.py', 'hermes_job_entrypoint.py', 'hermes_coordinator_runtime.py')


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--stage', type=Path, required=True)
    parser.add_argument('--image', required=True)
    args = parser.parse_args()
    helpers = runpy.run_path(str(args.stage / 'activate-basic-delegation.py'))
    require, run, atomic, encoded, idle = (helpers[k] for k in ('require','run','atomic','encoded','idle'))
    root, package = helpers['ROOT'], helpers['PACKAGE']
    require(os.geteuid() == 0 and socket.gethostname() == 'archived-worker.invalid', 'root_omarchy_required')
    require(args.stage.is_relative_to(root/'qualifications') and args.stage.resolve() == args.stage
            and args.stage.stat().st_uid == 0 and not args.stage.stat().st_mode & 0o022, 'protected_stage_required')
    require(re.fullmatch(r'sha256:[0-9a-f]{64}', args.image), 'immutable_image_required')
    inspected = json.loads(run('/usr/bin/docker','image','inspect',args.image))[0]
    require(inspected['Id'] == args.image and inspected['Os'] == 'linux' and inspected['Architecture'] == 'amd64', 'image_identity_mismatch')
    bindings = json.loads((args.stage/'source-binding.json').read_text())
    sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
    for name in MODULES:
        require(sha(package/name) == bindings['before'][name] and sha(args.stage/name) == bindings['after'][name], 'source_drift')
    config_paths = {name:Path('/etc/cloud-workbench')/(name+'.json') for name in ('api','worker')}
    originals = {name:path.read_bytes() for name,path in config_paths.items()}
    configs = {name:json.loads(raw) for name,raw in originals.items()}
    database = root/'control/state.db'
    registry_path = root/'environments/hermes-tasks-web-v2.db'
    backup = root/'operator-backups/web-delegation-20260918'
    require(not backup.exists() and not registry_path.exists(), 'web_activation_already_attempted')
    idle(database)
    legacy = run('systemctl','show','cloudd','--property=MainPID','--value')
    backup.mkdir(mode=0o700)
    receipt = {'host':'archived-worker.invalid','started_at':time.time(),'stage':'preparing','new_job_runs':0,
               'verification_runs':0,'image':args.image,'environment_version':VERSION,
               'environment_qualified':False,'authorization':'The operator requested internet and Chromium on tool containers',
               'source_sha256':bindings['after'],'backup':str(backup)}
    def record():atomic(backup/'deployment.json',encoded(receipt))
    record()
    try:
        for service in ('cloud-workbench-api','cloud-workbench-worker'):run('systemctl','stop',service)
        idle(database)
        for name,path in config_paths.items():
            require(path.read_bytes() == originals[name], 'config_drift')
            atomic(backup/(name+'.json'),originals[name])
        for name in MODULES:
            require(sha(package/name) == bindings['before'][name], 'source_drift')
            atomic(backup/name,(package/name).read_bytes())
        with sqlite3.connect(database.as_uri()+'?mode=ro',uri=True) as src, sqlite3.connect(backup/'state.db') as dst:src.backup(dst)
        (backup/'state.db').chmod(0o600)
        previous = Path(configs['worker']['environment_registry'])
        require(configs['api']['environment_registry'] == str(previous), 'registry_config_mismatch')
        with sqlite3.connect(previous.as_uri()+'?mode=ro',uri=True) as src, sqlite3.connect(registry_path) as dst:src.backup(dst)
        from cloudworkbench.environments import EnvironmentRegistry
        registry = EnvironmentRegistry(registry_path)
        old = registry.resolve('hermes-tasks','hermes-tasks-v1')
        manifest = copy.deepcopy(old['manifest'])
        manifest.update(version=VERSION,image_digest=args.image,network_profile=PROFILE)
        manifest['resources']['workspace_mib'] = 8192
        manifest['cli_versions']['playwright'] = '1.63.0'
        manifest['readiness_probes'] = [{'id':'browser-packages','argv':['/opt/hermes/venv/bin/python','-c','import playwright.sync_api'], 'timeout_seconds':30}]
        candidate = registry.register(manifest)
        require(candidate['qualified'] is False and candidate['qualification'] is None, 'unexpected_qualification')
        receipt['manifest_sha256'] = candidate['manifest_sha256']
        receipt['environment'] = candidate
        for name,cfg in configs.items():
            project = cfg['projects']['hermes-tasks']
            require(VERSION not in project['environment_versions'], 'environment_already_configured')
            project['environment_versions'].append(VERSION)
            project.setdefault('operator_approved_environments',{})[VERSION] = candidate['manifest_sha256']
            project.setdefault('environment_resources',{})[VERSION] = manifest['resources']
            cfg['environment_registry'] = str(registry_path)
        worker = configs['worker']
        worker['operator_approved_images'] = list(dict.fromkeys(worker.get('operator_approved_images',[])+[args.image]))
        worker['admission_memory_mib'] = 4096
        worker['admission_workspace_mib'] = 8192
        policy = worker['hermes_runtime']
        old_profile = policy['environment_network_profile']
        policy['environment_network_profiles'] = [old_profile,PROFILE]
        policy['tool_network_by_profile'] = {old_profile:'none',PROFILE:'bridge'}
        policy['tool_image_allowlist'] = [old['manifest']['image_digest'],args.image]
        receipt['stage'] = 'publication'
        record()
        for name in MODULES:atomic(package/name,(args.stage/name).read_bytes(),0o644)
        os.chown(registry_path,0,960);os.chmod(registry_path,0o640)
        atomic(registry_path.with_suffix('.approval.json'),encoded({'manifest_sha256':candidate['manifest_sha256'],
               'image':args.image,'operator_approved':True,'qualified':False,'tests_run':0,'profile':PROFILE}),0o640,960)
        for name,path in config_paths.items():atomic(path,encoded(configs[name]),0o640,960)
        for service in ('cloud-workbench-worker','cloud-workbench-api'):run('systemctl','start',service)
        receipt['services'] = {service:run('systemctl','is-active',service) for service in ('cloud-workbench-worker','cloud-workbench-api')}
        require(run('systemctl','show','cloudd','--property=MainPID','--value') == legacy, 'legacy_service_changed')
        receipt['stage'] = 'activated'
    except Exception as exc:
        receipt['error'] = str(exc) if type(exc) is RuntimeError else type(exc).__name__
        receipt['recovery'] = 'Inspect retained backup and partial publication; no blind rerun.'
        raise
    finally:
        receipt['finished_at'] = time.time();record()
    print(json.dumps({k:v for k,v in receipt.items() if k != 'environment'},indent=2))


if __name__ == '__main__':
    try:main()
    except Exception as exc:
        print(json.dumps({'activated':False,'error':str(exc) if type(exc) is RuntimeError else type(exc).__name__}));sys.exit(1)
