"""Synthetic relay process launched by trusted docker exec as numeric UID 1001."""

# Archived one-off operation; use the supported host installer instead.
raise SystemExit('Archived operation is disabled. See docs/AGENT-SETUP.md for supported installation.')

import hashlib,json,os,signal,threading,time
from pathlib import Path
from cloudworkbench.inference_relay import AttemptBinding,DispatchResult,InferenceRelay,WorkerDispatcher,WorkerSocketServer,make_loopback_service
from cloudworkbench.inference_service import ServiceResponse

assert os.getuid()==1001
root=Path('/run/relay');assert root.stat().st_uid==1001 and root.stat().st_mode&0o777==0o700
records=[]
def execute(body,cancel):
    assert body.get('model')=='claude-opus-4-6'
    found=any(m.get('role')=='tool' and 'SYNTHETIC_DISTINCT_UID_FILE' in str(m.get('content','')) for m in body.get('messages',[]))
    call={'id':'call_uid_fixture','type':'function','function':{'name':'read_file','arguments':'{"path":"/run/tool/fixture.txt"}'}}
    chunks=[{'role':'assistant'}, {'content':'SYNTHETIC_DISTINCT_UID_OK'} if found else {'tool_calls':[{'index':0,**call}]}]
    frames=[]
    for delta in chunks:
        frames.append({'id':'chatcmpl-uid','object':'chat.completion.chunk','created':1,'model':body['model'],'choices':[{'index':0,'delta':delta,'finish_reason':None}],'x_cloudworkbench':{'synthetic':True,'provider_calls':False}})
    frames.append({'id':'chatcmpl-uid','object':'chat.completion.chunk','created':1,'model':body['model'],'choices':[{'index':0,'delta':{},'finish_reason':'stop' if found else 'tool_calls'}]})
    data=b''.join(b'data: '+json.dumps(frame).encode()+b'\n\n' for frame in frames)+b'data: [DONE]\n\n'
    records.append({'tool_result_received':found,'uid':os.getuid(),'cgroup_measurement':{name:Path('/sys/fs/cgroup',name).read_text() for name in ('cgroup.procs','cpu.max','memory.max','pids.max')}})
    (root/'requests.json').write_text(json.dumps(records));(root/'requests.json').chmod(0o600)
    # Synthetic callback creates no provider resources. This is not provider cleanup qualification.
    return DispatchResult(ServiceResponse(200,'text/event-stream',data),True)

binding=AttemptBinding('uid-proof',1,hashlib.sha256(b'synthetic uid profile').hexdigest())
worker=WorkerDispatcher(journal_path=root/'worker.db',binding=binding,capability_sha256=hashlib.sha256(('c'*64).encode()).hexdigest(),authorize=lambda b:True,execute=execute)
uds=WorkerSocketServer(root/'worker.sock',worker,deadline_seconds=15)
thread=threading.Thread(target=uds.serve_forever,daemon=True);thread.start()
relay=InferenceRelay(socket_path=root/'worker.sock',journal_path=root/'relay.db',binding=binding,capability='c'*64,fence=lambda code:records.append({'fence':code}),deadline_seconds=15)
http=make_loopback_service(relay,address=('127.0.0.1',9876),capability_sha256=hashlib.sha256(('b'*64).encode()).hexdigest(),authorize=lambda:True,execution_timeout=20)
state={'uid':os.getuid(),'gid':os.getgid(),'pid':os.getpid(),'cgroup':Path('/proc/self/cgroup').read_text(),'status':{line.split(':')[0]:line.split(':',1)[1].strip() for line in Path('/proc/self/status').read_text().splitlines() if line.split(':')[0] in ('Uid','Gid','CapEff','CapBnd','NoNewPrivs','Seccomp','Seccomp_filters')},'journal_mode':oct((root/'relay.db').stat().st_mode&0o777),'journal_uid':(root/'relay.db').stat().st_uid,'synthetic_only':True}
(root/'ready.json').write_text(json.dumps(state));(root/'ready.json').chmod(0o600)
http.serve_forever()
