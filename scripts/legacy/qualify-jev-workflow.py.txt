#!/usr/bin/env python3
"""Run the five synthetic requests authorized by the operator; never dispatch agents."""

# Archived one-off operation; use the supported host installer instead.
raise SystemExit('Archived operation is disabled. See docs/AGENT-SETUP.md for supported installation.')

import hashlib
import json
import subprocess
import time
from pathlib import Path
from cloudworkbench.jev_client import JevClient
from cloudworkbench.workflow_routing import WORKFLOWS, select_workflow

CASES = (
    ('feature', 'Build CSV export for a sample inventory application, including automated tests.'),
    ('bug_fix', 'Fix a reproducible crash when the sample shopping cart is empty; preserve existing behavior otherwise.'),
    ('plan_review', 'Only adversarially review this proposed plan: replace all database writes at once, without a migration or rollback. Find gaps and risky assumptions. Do not implement it.'),
    ('code_review', 'Review this existing sample function for correctness and edge cases: def average(xs): return sum(xs) / len(xs). Do not modify the code.'),
    ('no_match', 'Please do the thing we discussed somewhere else. No task details are available.'),
)


def main():
    root = Path(__file__).resolve().parents[2]
    receipt = root/'evidence/jev-workflow-live-authorized-five.jsonl'
    # Exclusive creation prevents an accidental second batch under the same authorization.
    with receipt.open('x') as log:
        def record(row):
            log.write(json.dumps(row, allow_nan=False)+'\n'); log.flush()
            import os
            os.fsync(log.fileno())
        record({'kind':'authorization','max_requests':5,'synthetic_only':True,'dispatch_performed':False,
                'source_hashes':{n:hashlib.sha256((root/'src/cloudworkbench'/n).read_bytes()).hexdigest()
                                 for n in ('jev_client.py','workflow_routing.py','pstack_routing.py')}})
        auth = subprocess.run(['op','item','get','musuvrnksfwqbd4uveurdxck4m','--fields','credential','--reveal'],capture_output=True,text=True,timeout=60)
        if auth.returncode:
            record({'kind':'blocked','reason':'credential_unavailable'}); return 2
        client=JevClient(auth.stdout.strip()); del auth
        for ordinal,(expected,summary) in enumerate(CASES,1):
            record({'kind':'intent','ordinal':ordinal,'expected':expected,
                    'summary_sha256':hashlib.sha256(summary.encode()).hexdigest()})
            started=time.monotonic()
            try:
                result=select_workflow(summary,allowed_workflows=list(WORKFLOWS),client=client,external_allowed=True)
                data=result.receipt()
                record({'kind':'result','ordinal':ordinal,'expected':expected,
                        'matches_expected':result.workflow==expected if expected!='no_match' else result.reason=='no_match',
                        'elapsed_seconds':round(time.monotonic()-started,4),'result':data})
                print(json.dumps({'ordinal':ordinal,'expected':expected,'status':result.status,
                                  'workflow':result.workflow,'reason':result.reason,'elapsed_seconds':round(time.monotonic()-started,4)}),flush=True)
            except Exception:
                record({'kind':'error','ordinal':ordinal,'reason':'qualification_failed'})
                print('Qualification failed; no automatic retry.',flush=True); return 2
    return 0

if __name__=='__main__':raise SystemExit(main())
