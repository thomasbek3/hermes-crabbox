#!/usr/bin/env python3
"""Isolated real-Docker completion commit crash probe, without provider credentials."""

# Archived one-off operation; use the supported host installer instead.
raise SystemExit('Archived operation is disabled. See docs/AGENT-SETUP.md for supported installation.')

import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time
import uuid

from cloudworkbench.runner import Runner
from cloudworkbench.store import Store


def child(folder, phase):
    cfg = json.loads((folder / 'config.json').read_text())
    store = Store(folder / 'state.db', shared_group=True)
    runner = Runner(store, cfg)
    runner.capacity = lambda: 0
    if phase != 'resume':
        transition = store.transition

        def crash(identity, state, **fields):
            if state == 'completed':
                if phase == 'after':
                    transition(identity, state, **fields)
                os._exit(73)
            return transition(identity, state, **fields)

        store.transition = crash
    deadline = time.monotonic() + 80
    while time.monotonic() < deadline:
        runner.tick()
        with store._connect() as db:
            state = db.execute('SELECT state FROM attempts LIMIT 1').fetchone()[0]
        if state in ('completed', 'failed', 'cancelled', 'interrupted'):
            return
        time.sleep(.15)
    raise RuntimeError('qualification deadline exceeded')


def main():
    if len(sys.argv) == 4 and sys.argv[1] == '--child':
        child(Path(sys.argv[2]), sys.argv[3])
        return
    assert os.uname().nodename == 'archived-worker.invalid'
    assert os.getuid() != 0
    root = Path(sys.argv[1])
    root.mkdir(mode=0o2770, exist_ok=False)
    live = json.loads(Path('/etc/cloud-workbench/worker.json').read_text())
    fixture = json.loads(Path('/etc/cloud-workbench/qualification.json').read_text())
    receipts = []
    for phase in ('before', 'after'):
        folder = root / phase
        folder.mkdir(mode=0o2770)
        cfg = {'state_root': str(folder), 'input_root': str(folder / 'inputs'),
               'test_mode': True, 'capacity': 1, 'execution_seconds': 50,
               'projects': fixture['projects'],
               'runtime': {**live['runtime'], 'owner': 'completionprobe',
                           'network_enabled': False, 'workspace_mib': 64,
                           'memory_mib': 256, 'cpus': .5, 'pids': 64,
                           'approved_mount_roots': [str(folder)],
                           'approved_writable_mount_roots': [live['runtime']['root']]}}
        (folder / 'config.json').write_text(json.dumps(cfg))
        store = Store(folder / 'state.db', shared_group=True)
        owner = store.add_client('synthetic-completion', uuid.uuid4().hex * 2,
                                 ['submit', 'observe', 'retrieve'], ['sample-web'])
        created = store.create_session(owner, {'project_id': 'sample-web', 'goal': 'Fixture booking correction',
                                              'agent': 'fixture', 'environment_version': 'demo-v1'}, uuid.uuid4().hex)
        proc = subprocess.run([sys.executable, __file__, '--child', str(folder), phase],
                              stdout=subprocess.DEVNULL, stderr=subprocess.PIPE, timeout=100)
        if proc.returncode != 73:
            raise RuntimeError('crash child did not reach commit boundary: ' + str(proc.returncode) + ' ' + proc.stderr.decode()[-1200:])
        crashed = store.get_attempt(created['attempt_id'])
        assert crashed['state'] == ('verifying' if phase == 'before' else 'completed')
        events_before = store.events(owner, created['session_id'])
        subprocess.run([sys.executable, __file__, '--child', str(folder), 'resume'],
                       check=True, stdout=subprocess.DEVNULL, timeout=100)
        result = store.get_attempt(created['attempt_id'])
        events = store.events(owner, created['session_id'])
        assert result['state'] == 'completed' and result['outcome'] == 'verified'
        assert len([e for e in events if e['type'] == 'attempt.state' and e['payload'].get('state') == 'completed']) == 1
        assert not store.active_attempts()
        prior_running = [e for e in events_before if e['type'] == 'attempt.state' and e['payload'].get('state') == 'running']
        now_running = [e for e in events if e['type'] == 'attempt.state' and e['payload'].get('state') == 'running']
        assert len(prior_running) == len(now_running) == 1
        sidecar = folder / 'results' / (created['attempt_id'] + '.json')
        assert json.loads(sidecar.read_text()) == result['result']
        containers = subprocess.check_output(['/usr/bin/docker', 'ps', '--all', '--quiet',
                                             '--filter', 'label=io.cloudworkbench.owner=completionprobe'], text=True).strip()
        assert not containers
        receipts.append({'phase': phase, 'session_id': created['session_id'], 'attempt_id': created['attempt_id'],
                         'crash_exit': proc.returncode, 'state_at_crash': crashed['state'],
                         'final_state': result['state'], 'outcome': result['outcome'],
                         'execution_running_events': len(now_running), 'owned_containers': 0,
                         'result_sidecar_sha256': hashlib.sha256(sidecar.read_bytes()).hexdigest()})
    import cloudworkbench.runner as module
    print(json.dumps({'passed': True, 'host': os.uname().nodename, 'provider_called': False,
                      'image': live['runtime']['image'], 'runner_sha256': hashlib.sha256(Path(module.__file__).read_bytes()).hexdigest(),
                      'cases': receipts}, indent=2))


if __name__ == '__main__':
    main()
