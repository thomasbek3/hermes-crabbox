requests=[];sent_tool=False
class Handler(BaseHTTPRequestHandler):
 def log_message(self,*args):pass
 def do_POST(self):
  global sent_tool
  body=json.loads(self.rfile.read(int(self.headers['Content-Length'])))
  requests.append({'path':self.path,'model':body.get('model'),'stream':body.get('stream',False),
                   'tool_names':[tool.get('name') for tool in body.get('tools',[])],
                   'has_fixture_tool_result':'SYNTHETIC_FILE_CONTENT' in json.dumps(body.get('messages',[]))})
  if self.path.endswith('/count_tokens'):
   data=json.dumps({'input_tokens':32}).encode();self.send_response(200);self.send_header('Content-Type','application/json');self.end_headers();self.wfile.write(data);return
  if self.path.split('?')[0]!='/v1/messages':
   self.send_response(404);self.send_header('Content-Type','application/json');self.end_headers();self.wfile.write(b'{"error":"synthetic_endpoint_not_supported"}');return
  tool=not sent_tool and 'read_file' in requests[-1]['tool_names']
  if tool:sent_tool=True
  block={'type':'tool_use','id':'toolu_synthetic','name':'read_file','input':{'path':'/tmp/probe.txt'}} if tool else {'type':'text','text':'SYNTHETIC_FILE_TOOL_OK'}
  message={'id':'msg_synthetic','type':'message','role':'assistant','model':'synthetic-local-probe','content':[block],
           'stop_reason':'tool_use' if tool else 'end_turn','stop_sequence':None,'usage':{'input_tokens':32,'output_tokens':10}}
  self.send_response(200)
  if body.get('stream'):
   self.send_header('Content-Type','text/event-stream');self.end_headers()
   def emit(kind,data):self.wfile.write(('event: '+kind+'\ndata: '+json.dumps(data)+'\n\n').encode());self.wfile.flush()
   emit('message_start',{'type':'message_start','message':{**message,'content':[],'stop_reason':None,'usage':{'input_tokens':32,'output_tokens':0}}})
   initial={**block,'input':{}} if tool else {'type':'text','text':''}
   emit('content_block_start',{'type':'content_block_start','index':0,'content_block':initial})
   delta={'type':'input_json_delta','partial_json':json.dumps(block['input'])} if tool else {'type':'text_delta','text':block['text']}
   emit('content_block_delta',{'type':'content_block_delta','index':0,'delta':delta})
   emit('content_block_stop',{'type':'content_block_stop','index':0})
   emit('message_delta',{'type':'message_delta','delta':{'stop_reason':message['stop_reason'],'stop_sequence':None},'usage':{'output_tokens':10}})
   emit('message_stop',{'type':'message_stop'})
  else:
   self.send_header('Content-Type','application/json');self.end_headers();self.wfile.write(json.dumps(message).encode())
